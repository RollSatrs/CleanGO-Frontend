"use client";

import { useState } from "react";
import Image from "next/image";
import { useRouter } from "next/navigation";

import {
  AnimatePresence,
  motion,
  useReducedMotion,
} from "motion/react";

import { ArrowLeft, ArrowRight } from "lucide-react";

type OnboardingSlide = {
  title: string;
  description: string;
  image: string;
  imageAlt: string;
};

const slides: OnboardingSlide[] = [
  {
    title: "Чистый дом\nбез забот",
    description:
      "Закажите уборку, а мы подберём проверенного клинера",
    image: "/images/cleango/living-room.png",
    imageAlt: "Уютное кресло и комнатное растение",
  },
  {
    title: "Быстро,\nудобно, безопасно",
    description:
      "Прозрачная стоимость, надёжные исполнители и контроль качества",
    image: "/images/cleango/onboarding-cleaning-brush.png",
    imageAlt: "3D-щётка для уборки CleanGo",
  },
];

export default function OnboardingPage() {
  const router = useRouter();
  const reduceMotion = useReducedMotion();

  const [step, setStep] = useState(0);

  const current = slides[step];

  function handleBack() {
    if (step > 0) {
      setStep((previous) => previous - 1);
    } else {
      router.push("/welcome");
    }
  }

  function handleNext() {
    if (step < slides.length - 1) {
      setStep((previous) => previous + 1);
    } else {
      router.push("/");
    }
  }

  function handleSkip() {
    router.push("/");
  }

  const appear = (delay = 0) => ({
    initial: {
      opacity: 0,
      y: reduceMotion ? 0 : 12,
    },
    animate: {
      opacity: 1,
      y: 0,
    },
    transition: {
      duration: reduceMotion ? 0 : 0.45,
      delay: reduceMotion ? 0 : delay,
      ease: [0.22, 1, 0.36, 1] as const,
    },
  });

  return (
    <main className="min-h-[100dvh] bg-[#F1FBFC] font-sans">
      <section
        className="
          cleango-screen
          mx-auto flex w-full
          max-w-[520px] flex-col

              active:opacity-60

          </button>

          <button
            type="button"
            onClick={handleSkip}
            className="
              text-sm font-medium
              text-[#8392A2]
              transition-opacity
              active:opacity-60
            "
          >
            Пропустить
          </button>
        </motion.header>

        {/* Центральная часть */}
        <div className="flex min-h-0 flex-1 flex-col justify-center">
          <AnimatePresence mode="wait" initial={false}>
            <motion.div
              key={step}
              initial={{
                opacity: 0,
                x: reduceMotion ? 0 : 14,
              }}
              animate={{
                opacity: 1,
                x: 0,
              }}

                  alt={current.imageAlt}

                  priority
                  sizes="(max-width: 520px) 90vw, 350px"
                  className="
                    h-auto max-h-full
                    w-auto max-w-full
                    object-contain

                />

  if (typeof window === "undefined") {

function readUrlData(): string {
  if (typeof window === "undefined") {
    return "";
  }

  const query = new URLSearchParams(
    window.location.search,
  );

  const queryData = query.get("tgWebAppData");

  if (queryData) {
    return queryData;
  }

  const hash = window.location.hash.replace(
    /^#/,
    "",
  );

  const hashParams = new URLSearchParams(hash);

  return hashParams.get("tgWebAppData") ?? "";
}

  try {

      ? params.initDataRaw
      : "";
  } catch {
    return "";
  }
}

// ==========================================
// SAFE FIELD DIAGNOSTICS
// ==========================================

function inspectInitData(data: string) {
  if (!data) {
    return {
      fields: [] as string[],
      hasHash: false,
      hasUser: false,
      hasAuthDate: false,
      hasSignature: false,
      duplicateKeys: [] as string[],
    };
  }

  const params = new URLSearchParams(data);
  const fields = [...params.keys()];

  const counts = new Map<string, number>();

  for (const key of fields) {
    counts.set(
      key,
      (counts.get(key) ?? 0) + 1,
    );
  }

  return {
    fields,
    hasHash: Boolean(params.get("hash")),
    hasUser: Boolean(params.get("user")),
    hasAuthDate: Boolean(
      params.get("auth_date"),
    ),
    hasSignature: Boolean(
      params.get("signature"),
    ),
    duplicateKeys: [
      ...counts.entries(),
    ]
      .filter(([, count]) => count > 1)
      .map(([key]) => key),
  };
}

// AUTH PROVIDER

export function AuthProvider({
  children,
}: {
  children: ReactNode;
}) {
  const [status, setStatus] =
    useState<AuthStatus>("loading");

  const [user, setUser] =
    useState<AuthUser | null>(null);

  const [token, setToken] =
    useState<string | null>(null);

  const [attempt, setAttempt] = useState(0);

  const [diagnostics, setDiagnostics] =
    useState<Diagnostics>(
      initialDiagnostics,
    );

  const retry = useCallback(() => {
    setAttempt((value) => value + 1);
  }, []);

  useEffect(() => {
    let cancelled = false;

    const update = (
      key: keyof Diagnostics,
      checkStatus: CheckStatus,
      detail: string,
    ) => {
      if (cancelled) {
        return;
      }

      setDiagnostics((previous) => ({
        ...previous,
        [key]: {
          status: checkStatus,
          detail,
        },
      }));
    };

    async function authenticate() {
      setStatus("loading");
      setDiagnostics(
        initialDiagnostics(),
      );

      try {
        // ==================================
        // 1. TELEGRAM ENVIRONMENT
        // ==================================

        const insideTelegram =
          await isTMA();

        if (cancelled) {
          return;
        }

        update(
          "environment",
          insideTelegram
            ? "success"
            : "failed",
          insideTelegram
            ? "Telegram обнаружен"
            : "Приложение открыто вне Telegram",
        );

        if (!insideTelegram) {
          setStatus("outside-telegram");
          return;
        }

        // ==================================
        // 2. WEBAPP
        // ==================================

        const telegramWindow =
          window as TelegramWindow;

        const webAppExists = Boolean(
          telegramWindow.Telegram?.WebApp,
        );

        const webAppData =
          readWebAppData();

        update(
          "webApp",
          webAppData
            ? "success"
            : "failed",
          !webAppExists
            ? "Объект WebApp отсутствует"
            : webAppData
              ? "WebApp доступен, initData получены"
              : "WebApp доступен, но initData пустые",
        );

        // ==================================
        // 3. URL DATA
        // ==================================

        const urlData = readUrlData();

        const inspected =
          inspectInitData(urlData);

        const fieldsText =
          inspected.fields.length > 0
            ? inspected.fields.join(", ")
            : "нет полей";

        const validationText = [
          `hash: ${inspected.hasHash ? "есть" : "нет"}`,
          `user: ${inspected.hasUser ? "есть" : "нет"}`,
          `auth_date: ${inspected.hasAuthDate ? "есть" : "нет"}`,
          `signature: ${inspected.hasSignature ? "есть" : "нет"}`,
        ].join("; ");

        const duplicateText =
          inspected.duplicateKeys.length > 0
            ? `; повторяющиеся поля: ${inspected.duplicateKeys.join(", ")}`
            : "";

        update(
          "urlData",
          urlData
            ? "success"
            : "failed",
          urlData
            ? `tgWebAppData обнаружен. Поля: ${fieldsText}. Проверка: ${validationText}${duplicateText}`
            : "tgWebAppData не найден",
        );

        // ==================================
        // 4. SDK DATA
        // ==================================

        const sdkData = readSdkData();

        update(
          "sdkData",
          sdkData
            ? "success"
            : "failed",
          sdkData
            ? "SDK получил initDataRaw"
            : "SDK не получил initDataRaw",
        );

        // ==================================
        // 5. CHOOSE INIT DATA
        // ==================================

        const initData =
          webAppData ||
          sdkData ||
          urlData;

        if (!initData) {
          update(
            "authorization",
            "failed",
            "Данные авторизации Telegram отсутствуют",
          );

          setStatus("error");
          return;
        }

        // ==================================
        // 6. EXISTING SESSION
        // ==================================

        const existingToken =
          sessionStorage.getItem(
            TOKEN_KEY,
          );

        if (existingToken) {
          try {
            const response = await fetch(
              `${API_URL}/auth/me`,
              {
                headers: {
                  Authorization:
                    `Bearer ${existingToken}`,
                },
                cache: "no-store",
              },
            );

            if (response.ok) {
              const currentUser =
                (await response.json()) as AuthUser;

              if (cancelled) {
                return;
              }

              update(
                "backend",
                "success",
                `HTTP ${response.status}`,
              );

              update(
                "authorization",
                "success",
                "Сессия подтверждена",
              );

              setUser(currentUser);
              setToken(existingToken);
              setStatus(
                "authenticated",
              );

              return;
            }
          } catch {
            // Попробуем Telegram login.
          }

          sessionStorage.removeItem(
            TOKEN_KEY,
          );
        }

        // ==================================
        // 7. TELEGRAM LOGIN
        // ==================================

        const response = await fetch(
          `${API_URL}/auth/telegram`,
          {
            method: "POST",
            headers: {
              "Content-Type":
                "application/json",
            },
            body: JSON.stringify({
              initData,
            }),
            cache: "no-store",
          },
        );

        if (cancelled) {
          return;
        }

        update(
          "backend",
          "success",
          `HTTP ${response.status}`,
        );

        if (!response.ok) {
          const errorBody =
            await response
              .json()
              .catch(() => null);

          const serverMessage =
            errorBody &&
            typeof errorBody.message ===
              "string"
              ? errorBody.message
              : "Неизвестная ошибка";

          update(
            "authorization",
            "failed",
            `HTTP ${response.status}: ${serverMessage}`,
          );

          setStatus("error");
          return;
        }

        // ==================================
        // 8. JWT RESPONSE
        // ==================================

        const result =
          (await response.json()) as AuthResponse;

        if (
          !result.accessToken ||
          !result.user?.id
        ) {
          update(
            "authorization",
            "failed",
            "Некорректный ответ сервера",
          );

          setStatus("error");
          return;
        }

        sessionStorage.setItem(
          TOKEN_KEY,
          result.accessToken,
        );

        setUser(result.user);
        setToken(
          result.accessToken,
        );

        update(
          "authorization",
          "success",
          "JWT получен",
        );

        setStatus("authenticated");
      } catch (error) {
        if (cancelled) {
          return;
        }

        console.error(
          "CleanGo auth diagnostic error:",
          error,
        );

        update(
          "backend",
          "failed",
          "Ошибка выполнения запроса",
        );

        setStatus("error");
      }
    }

    void authenticate();

    return () => {
      cancelled = true;
    };
  }, [attempt]);

  return (
    <AuthContext.Provider
      value={{
        user,
        token,
        status,
        retry,
      }}
    >
      {status === "authenticated" ? (
        children
      ) : status === "outside-telegram" ? (
        <main className="flex min-h-dvh items-center justify-center bg-slate-50 px-5">
          <div className="rounded-3xl bg-white p-8 text-center shadow-lg">
            <h1 className="text-xl font-bold">
              Откройте CleanGo в Telegram
            </h1>

            <a
              href="https://t.me/cleaninggoo_bot"
              className="mt-5 inline-block rounded-xl bg-sky-500 px-6 py-3 font-semibold text-white"
            >
              Перейти к боту
            </a>
          </div>
        </main>
      ) : (
        <DiagnosticScreen
          diagnostics={diagnostics}
          retry={retry}
        />
      )}
    </AuthContext.Provider>
  );
}

// ==========================================
// USE AUTH
// ==========================================

export function useAuth() {
  const context = useContext(
    AuthContext,
  );

  if (!context) {
    throw new Error(
      "useAuth должен использоваться внутри AuthProvider",
    );
  }

  return context;
}
